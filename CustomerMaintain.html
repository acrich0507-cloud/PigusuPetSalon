// CustomerMaintain.gs // ========================================== // 1. 負責「新增/修改」會員資料 (Upsert) // ========================================== function handleInsertUser(payload) { const url = `${APP_CONFIG.supabase.url}/rest/v1/Customer`; const body = { U_ID: payload.uid, U_LineName: payload.lineName || null, U_Name: payload.name, U_Six: payload.six || null, U_Phone: payload.phone, U_Addr: payload.addr || null, U_Email: payload.email || null, U_BirthDay: payload.birthday || null, U_Memo: payload.memo || null, U_Marketing: payload.marketing, U_Flag: true, Update_Time: new Date().toISOString() }; const headers = getSupabaseHeaders(); headers['Prefer'] = 'resolution=merge-duplicates'; const options = { method: 'post', headers: headers, payload: JSON.stringify(body), muteHttpExceptions: true }; const response = UrlFetchApp.fetch(url, options); const responseCode = response.getResponseCode(); const responseText = response.getContentText(); if (responseCode === 201 || responseCode === 200 || responseCode === 204) { return createJsonResponse({ success: true }); } else { return createJsonResponse({ success: false, error: responseText }); } } // ========================================== // 2. 🌟 負責「查詢」單一會員完整資料 (你剛剛漏掉的這段) // ========================================== function handleGetUserData(uid) { if (!uid) { return createJsonResponse({ success: false, error: '缺少 UID' }); } const url = `${APP_CONFIG.supabase.url}/rest/v1/Customer?U_ID=eq.${uid}&select=*`; const options = { method: 'get', headers: getSupabaseHeaders(), muteHttpExceptions: true }; try { const response = UrlFetchApp.fetch(url, options); const data = JSON.parse(response.getContentText()); if (data.error || data.code) { return createJsonResponse({ success: false, error: data.message }); } if (Array.isArray(data) && data.length > 0) { return createJsonResponse({ success: true, data: data[0] }); } else { return createJsonResponse({ success: true, data: null }); } } catch (error) { return createJsonResponse({ success: false, error: error.toString() }); } }
